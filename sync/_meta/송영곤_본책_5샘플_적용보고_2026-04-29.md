#### 2.2.2 본문 OCR 자동 batch Python 스크립트 (사용자 실행)

```python
# 전제: 백업 완료 (위 PowerShell 실행 후)

import re
from pathlib import Path

SRC_DIR = Path(r"H:\내 드라이브\sync\_교재원문\민법\송영곤_논점민법_본책")
EXCLUDE = {"_교재목차.md"}

# 안전 패턴 (CLAUDE.md #34 anchor 보존)
PATTERNS = [
    # 사건번호 끝 각주 마커 (가장 안전)
    (re.compile(r'(\d+다\d+)\[\^\d+\]'), r'\1', "사건번호 끝 각주"),
    (re.compile(r'(\d+민[가-힣]+\d+)\[\^\d+\]'), r'\1', "민재항 각주"),
    # 페이지 헤더 본문 단독 라인 (제N부 ...)
    (re.compile(r'^\d{2,4}\s+제[O@©0o]?부\s+.+$', re.MULTILINE), '', "페이지 헤더 (숫자+제N부)"),
    (re.compile(r'^제\d+장.+\s+\d{1,4}\s*$', re.MULTILINE), '', "페이지 헤더 (제N장)"),
]

# 위험 패턴 (보류 — LLM 검증 필요)
SKIP = [
    "callout 안 페이지 헤더",
    "한자 깨짐 (anchor 보호)",
    "OCR 마커 깨짐 (IW日!, I@E1 등)",
    "본문 단독 숫자 라인",
]

def normalize(content: str) -> tuple[str, dict[str, int]]:
    counts = {}
    for pat, repl, name in PATTERNS:
        new_content, n = pat.subn(repl, content)
        counts[name] = n
        content = new_content
    # 빈 라인 연속 정리 (3개 이상 → 2개)
    content = re.sub(r'\n{3,}', '\n\n', content)
    return content, counts

def main():
    files = [f for f in SRC_DIR.glob("*.md") if f.name not in EXCLUDE and not f.name.startswith("_")]
    print(f"대상: {len(files)}개 파일")
    total = {name: 0 for _, _, name in PATTERNS}
    for f in files:
        content = f.read_text(encoding="utf-8")
        new_content, counts = normalize(content)
        if new_content != content:
            f.write_text(new_content, encoding="utf-8")
            for k, v in counts.items():
                total[k] += v
            print(f"  {f.name}: {sum(counts.values())}건")
    print("\n패턴별 합계:")
    for k, v in total.items():
        print(f"  {k}: {v}건")
    print(f"\n보류 패턴 (LLM 직접 처리 권장): {SKIP}")

if __name__ == "__main__":
    main()
```

2. 위 Python 스크립트를 `H:\내 드라이브\5.기타\scripts\normalize_송영곤_B형_본문.py`에 저장

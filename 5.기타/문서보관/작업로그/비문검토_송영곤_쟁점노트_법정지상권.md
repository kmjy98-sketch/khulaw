(re.compile(r'(\d{2,4})C\^(\d{3,6})'), r'\g<1>다\g<2>'),  # C^ → 다
(re.compile(r'(\d{2,4})P(\d{3,6})'), r'\g<1>다\g<2>'),    # P → 다
(re.compile(r'대EKM\)'), '대판(전합)'),
